<code>

man console_codes

priklad nastaveni promptu:

export PS1="[c:$?]$PWD> " .. $? vystupni kod posledni procedury

export $TMOUT=240 .. odpojeni po 4 minutach

for i in `seq 1 8`; do echo $i; done
</code>
